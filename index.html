function waitFor(test, label, ms = 15000) {

  return new Promise((resolve, reject) => {

    const start = Date.now();

    (function tick() {

      try { if (test()) return resolve(); } catch (e) {}

      if (Date.now() - start > ms) return reject(new Error(label));

      setTimeout(tick, 50);

    })();

  });

}

/* ====================================================================

const gcd = (a, b) => b === 0 ? Math.abs(a) : gcd(b, a % b);

const ri = (a, b) => a + Math.floor(Math.random() * (b - a + 1));

const pick = arr => arr[Math.floor(Math.random() * arr.length)];

const chance = p => Math.random() < p;

const M = (c, x = 0, y = 0, z = 0) => ({ coeff: c, x, y, z });

const clr = p => p.filter(t => t.coeff !== 0);

 

/* ====================================================================

   POLYNOMIAL HELPERS (kept for reduce-type problems)

   ==================================================================== */

function srtP(p) {

  return [...p].sort((a, b) => {

    const da = (a.x || 0) + (a.y || 0) + (a.z || 0);

    const db = (b.x || 0) + (b.y || 0) + (b.z || 0);

    if (da !== db) return db - da;

    if ((a.x || 0) !== (b.x || 0)) return (b.x || 0) - (a.x || 0);

    return (b.y || 0) - (a.y || 0);

  });

}

 

// Monomial → LaTeX

function mTeX(m) {

  const c = m.coeff || 1;

  let v = '';

  for (const k of ['x', 'y', 'z']) {

    const p = m[k] || 0;

    if (p === 1) v += k;

    else if (p >= 2) v += `${k}^{${p}}`;

  }

  if (!v) return String(c);

  if (c === 1) return v;

  if (c === -1) return '-' + v;

  return String(c) + v;

}

 

// Polynomial → LaTeX

function pTeX(poly) {

  if (!poly || !poly.length) return '0';

  const sp = srtP(clr(poly));

  if (!sp.length) return '0';

  let s = '';

  for (let i = 0; i < sp.length; i++) {

    const t = sp[i];

    const abs = { ...t, coeff: Math.abs(t.coeff) };

    const ts = mTeX(abs);

    if (i === 0) s += (t.coeff < 0 ? '-' : '') + ts;

    else s += (t.coeff < 0 ? ' - ' : ' + ') + ts;

  }

  return s;

}

 

function fracTeX(num, den) {

  return `\\frac{${pTeX(num)}}{${pTeX(den)}}`;

}

 

function mGCD(poly) {

  if (!poly || !poly.length) return M(1);

  let g = 0, mx = 99, my = 99, mz = 99;

  for (const t of poly) {

    g = gcd(g, Math.abs(t.coeff || 1));

    mx = Math.min(mx, t.x || 0);

    my = Math.min(my, t.y || 0);

    mz = Math.min(mz, t.z || 0);

  }

  return M(g || 1, mx === 99 ? 0 : mx, my === 99 ? 0 : my, mz === 99 ? 0 : mz);

}

 

function cFac(np, dp) {

  const ng = mGCD(np), dg = mGCD(dp);

  return M(gcd(ng.coeff, dg.coeff), Math.min(ng.x, dg.x), Math.min(ng.y, dg.y), Math.min(ng.z, dg.z));

}

   Each returns: { kind, problemLatex, simplifiedLatex, hint }

// --- Reduce-type: single fraction with cancellable common factor ---

function genReduce() {

  const type = ri(1, 6);

  let Q;

 

  if (type === 1) {

    let a1, a0, d;

    do { a1 = ri(1,4); a0 = ri(-3,4); if (!a0) a0 = ri(1,3); d = ri(2,5); }

    while (gcd(gcd(a1, Math.abs(a0)), d) > 1);

    const k = ri(2,4);

    Q = { qn: clr([M(k*a1,1), M(k*a0)]), qd: clr([M(k*d)]),

          sn: clr([M(a1,1), M(a0)]), sd: clr([M(d)]) };

  } else if (type === 2) {

    let b1, b0, c;

    do { b1 = ri(1,4); b0 = ri(-3,4); if (!b0) b0 = ri(1,3); c = ri(1,5); }

    while (gcd(gcd(b1, Math.abs(b0)), c) > 1);

    const k = ri(2,4);

    Q = { qn: clr([M(k*c)]), qd: clr([M(k*b1,1), M(k*b0)]),

          sn: clr([M(c)]), sd: clr([M(b1,1), M(b0)]) };

  } else if (type === 3) {

    let a1, a0, b;

    do { a1 = ri(1,4); a0 = ri(-3,4); if (!a0) a0 = ri(1,3); b = ri(1,4); }

    while (gcd(gcd(a1, Math.abs(a0)), b) > 1);

    const k = ri(2,3);

    Q = { qn: clr([M(k*a1,2), M(k*a0,1)]), qd: clr([M(k*b,1)]),

          sn: clr([M(a1,1), M(a0)]), sd: clr([M(b)]) };

  } else if (type === 4) {

    let a1, a0, d;

    do { a1 = ri(1,3); a0 = ri(-2,3); if (!a0) a0 = ri(1,2); d = ri(2,4); }

    while (gcd(gcd(a1, Math.abs(a0)), d) > 1);

    const k = ri(2,3);

    Q = { qn: clr([M(k*a1,1,0), M(k*a0,0,1)]), qd: clr([M(k*d)]),

          sn: clr([M(a1,1), M(a0,0,1)]), sd: clr([M(d)]) };

  } else if (type === 5) {

    let a1, a0, b;

    do { a1 = ri(1,3); a0 = ri(-2,3); if (!a0) a0 = ri(1,2); b = ri(1,3); }

    while (gcd(gcd(a1, Math.abs(a0)), b) > 1);

    const k = ri(2,3);

    Q = { qn: clr([M(k*a1,1,1), M(k*a0,0,2)]), qd: clr([M(k*b,0,1)]),

          sn: clr([M(a1,1), M(a0,0,1)]), sd: clr([M(b)]) };

  } else {

    let a1, a0, b;

    do { a1 = ri(1,3); a0 = ri(-2,3); if (!a0) a0 = ri(1,2); b = ri(1,3); }

    while (gcd(gcd(a1, Math.abs(a0)), b) > 1);

    const k = ri(2,3);

    Q = { qn: clr([M(k*a1,2,0), M(k*a0,1,1)]), qd: clr([M(k*b,1,0)]),

          sn: clr([M(a1,1), M(a0,0,1)]), sd: clr([M(b)]) };

  }

 

  const c = cFac(Q.qn, Q.qd);

  return {

    kind: 'reduce',

    problemLatex: fracTeX(Q.qn, Q.qd),

    simplifiedLatex: fracTeX(Q.sn, Q.sd),

    hint: `The numerator and denominator share a common factor of $${mTeX(c)}$. Divide both by it.`

  };

}

 

// --- Simple monomial generator for two-fraction problems ---

// Produces things like 2, 3x, 5y, xy, 2x^2  -- always positive coefficient

function randMono(vars, { allowConst = true } = {}) {

  const useVar = allowConst ? chance(0.75) : true;

  const coef = ri(1, 6);

  if (!useVar) return M(coef);

  if (vars.length >= 2 && chance(0.35)) {

    // product like xy or 2xy

    return M(coef, 1, 1);

  }

  const v = pick(vars);

  const p = chance(0.25) ? 2 : 1;

  if (v === 'x') return M(coef, p, 0);

  if (v === 'y') return M(coef, 0, p);

  return M(coef);

}

 

// Returns { n: monomial, d: monomial } where d ≠ 1 and n ≠ d

function randSimpleFrac(vars) {

  let n, d, tries = 0;

  do {

    n = randMono(vars);

    d = randMono(vars, { allowConst: true });

    tries++;

    if (tries > 30) break;

  } while (

    (d.coeff === 1 && d.x === 0 && d.y === 0 && d.z === 0) || // denom = 1

    (n.coeff === d.coeff && n.x === d.x && n.y === d.y && n.z === d.z) // same

  );

  return { n: [n], d: [d] };

}

 

function chooseVars() {

  // Mix of single-var and two-var problems

  return chance(0.55) ? ['x', 'y'] : ['x'];

}

 

// --- Two fractions combined: a/b OP c/d ---

function genTwoFrac() {

  const vars = chooseVars();

  const op = pick(['+', '-', 'mul', 'div']);

  const f1 = randSimpleFrac(vars);

  const f2 = randSimpleFrac(vars);

 

  const f1Tex = fracTeX(f1.n, f1.d);

  const f2Tex = fracTeX(f2.n, f2.d);

 

  let opTex, problemLatex, hint;

  if (op === '+') {

    opTex = '+';

    problemLatex = `${f1Tex} ${opTex} ${f2Tex}`;

    hint = 'Rewrite both fractions with a common denominator, then add the numerators.';

  } else if (op === '-') {

    opTex = '-';

    problemLatex = `${f1Tex} ${opTex} ${f2Tex}`;

    hint = 'Rewrite both fractions with a common denominator, then subtract the numerators.';

  } else if (op === 'mul') {

    opTex = '\\cdot';

    problemLatex = `${f1Tex} ${opTex} ${f2Tex}`;

    hint = 'Multiply the numerators together and the denominators together, then simplify.';

  } else {

    opTex = ':';

    problemLatex = `${f1Tex} ${opTex} ${f2Tex}`;

    hint = 'Dividing by a fraction is multiplying by its reciprocal.';

  }

 

  const simplifiedLatex = simplifyViaNerdamer(problemLatex);

  return { kind: 'two-frac', problemLatex, simplifiedLatex, hint };

}

 

// --- Complex fraction: fraction inside numerator and/or denominator ---

function genComplex() {

  const vars = chooseVars();

  const variant = ri(0, 2);

  const inner1 = randSimpleFrac(vars);

  let problemLatex, hint;

 

  if (variant === 0) {

    // (a/b) / c

    const c = randMono(vars, { allowConst: true });

    problemLatex = `\\frac{${fracTeX(inner1.n, inner1.d)}}{${mTeX(c)}}`;

    hint = 'Dividing by something is multiplying by its reciprocal — turn the bottom into a multiplier.';

  } else if (variant === 1) {

    // c / (a/b)

    const c = randMono(vars, { allowConst: true });

    problemLatex = `\\frac{${mTeX(c)}}{${fracTeX(inner1.n, inner1.d)}}`;

    hint = 'A division by a fraction equals multiplication by its reciprocal: flip the inner fraction.';

  } else {

    // (a/b) / (c/d)

    const inner2 = randSimpleFrac(vars);

    problemLatex = `\\frac{${fracTeX(inner1.n, inner1.d)}}{${fracTeX(inner2.n, inner2.d)}}`;

    hint = 'A fraction over a fraction equals the top times the reciprocal of the bottom.';

  }

 

  const simplifiedLatex = simplifyViaNerdamer(problemLatex);

  return { kind: 'complex', problemLatex, simplifiedLatex, hint };

}

 

// Pick a problem type, then generate

function genQ() {

  const roll = Math.random();

  if (roll < 0.45) return genReduce();

  if (roll < 0.80) return genTwoFrac();

  return genComplex();

}

 

/* ====================================================================

   NERDAMER BRIDGE — convert LaTeX and check equivalence

   ==================================================================== */

function manualLatexToNerdamer(latex) {

  let s = latex;

  s = s.replace(/\\left/g, '').replace(/\\right/g, '');

  s = s.replace(/\\cdot/g, '*').replace(/\\times/g, '*').replace(/\\div/g, '/');

  s = s.replace(/\\,|\\;|\\!/g, '');

  s = s.replace(/:/g, '/');

  let safety = 40;

  while (/\\frac\s*\{/.test(s) && safety-- > 0) {

    s = s.replace(/\\frac\s*\{([^{}]*)\}\s*\{([^{}]*)\}/g, '(($1)/($2))');

  }

  s = s.replace(/\^\s*\{([^{}]+)\}/g, '^($1)');

  s = s.replace(/\s+/g, '');

  return s;

}

 

function toNerdamerExpr(latex) {

  if (window.nerdamer && typeof window.nerdamer.convertFromLaTeX === 'function') {

    try {

      const sym = window.nerdamer.convertFromLaTeX(latex);

      return sym.text();

    } catch (e) {}

  }

  return manualLatexToNerdamer(latex);

}

 

function simplifyViaNerdamer(latex) {

  if (!window.nerdamer) return latex;

  try {

    const expr = toNerdamerExpr(latex);

    const candidates = [];

    const safe = fn => { try { const r = fn(); if (r) candidates.push(r.toTeX()); } catch (e) {} };

    safe(() => nerdamer(`simplify(${expr})`));

    safe(() => nerdamer(expr).expand());

    safe(() => nerdamer(`simplify(${nerdamer(expr).expand().text()})`));

    safe(() => nerdamer(expr));

    if (!candidates.length) return latex;

    candidates.sort((a, b) => a.length - b.length || a.localeCompare(b));

    return candidates[0];

  } catch (e) { return latex; }

}

 

function canonicalize(latex) {

  if (!window.nerdamer) return latex.replace(/\s+/g, '');

  try {

    return nerdamer(toNerdamerExpr(latex)).text();

  } catch (e) { return latex.replace(/\s+/g, ''); }

}

 

function checkEquivalent(latexA, latexB) {

  if (!window.nerdamer) return { ok: false, reason: 'no-nerdamer' };

  let exprA, exprB;

  try {

    exprA = toNerdamerExpr(latexA);

    exprB = toNerdamerExpr(latexB);

  } catch (e) { return { ok: false, reason: 'parse' }; }

  if (!exprB || exprB.trim() === '') return { ok: false, reason: 'empty' };

 

  let vars;

  try {

    const v1 = nerdamer(exprA).variables() || [];

    const v2 = nerdamer(exprB).variables() || [];

    vars = [...new Set([...v1, ...v2])];

  } catch (e) { return { ok: false, reason: 'parse' }; }

 

  if (vars.length === 0) {

    try {

      const a = parseFloat(nerdamer(exprA).evaluate().text('decimals'));

      const b = parseFloat(nerdamer(exprB).evaluate().text('decimals'));

      if (!isFinite(a) || !isFinite(b)) return { ok: false, reason: 'undefined' };

      return { ok: Math.abs(a - b) < 1e-9 * Math.max(1, Math.abs(a)), reason: 'numeric' };

    } catch (e) { return { ok: false, reason: 'eval' }; }

  }

 

  const TRIALS = 25;

  let evaluated = 0;

  for (let i = 0; i < TRIALS * 2 && evaluated < TRIALS; i++) {

    const subs = {};

    vars.forEach(v => {

      const base = 1.7 + Math.random() * 5;

      subs[v] = chance(0.5) ? base : -base;

    });

    let a, b;

    try {

      a = parseFloat(nerdamer(exprA, subs).evaluate().text('decimals'));

      b = parseFloat(nerdamer(exprB, subs).evaluate().text('decimals'));

    } catch (e) { continue; }

    if (!isFinite(a) || !isFinite(b)) continue;

    evaluated++;

    const scale = Math.max(1, Math.abs(a), Math.abs(b));

    if (Math.abs(a - b) > 1e-6 * scale) return { ok: false, reason: 'differs' };

  }

  if (evaluated < 5) return { ok: false, reason: 'unstable' };

  return { ok: true, reason: 'sampled' };

}

 

function isSimplifiedForm(studentLatex, simplifiedLatex) {

  // Canonical text comparison via nerdamer

  try {

    const cs = canonicalize(studentLatex);

    const ct = canonicalize(simplifiedLatex);

    if (cs === ct) return true;

  } catch (e) {}

  // Fallback heuristic: length within tolerance and equivalent

  const a = studentLatex.replace(/\s+/g, '').length;

  const b = simplifiedLatex.replace(/\s+/g, '').length;

  return a <= b + 1;

}

/* ---- Rendering ---- */

function addRewrite(latex, status) {

  const wrap = document.getElementById('rewrites');

  const el = document.createElement('div');

  el.className = 'rewrite ' + status;

  const mark = status === 'correct' ? '✓' : status === 'incorrect' ? '✕' : '★';

  const tag = status === 'correct' ? 'equivalent'

            : status === 'incorrect' ? 'not equivalent'

            : 'simplified form';

  el.innerHTML = `

    <div class="mark" aria-hidden="true">${mark}</div>

    <div class="math">${renderLatex(latex)}</div>

    <div class="tag">${tag}</div>

  `;

  wrap.appendChild(el);

  el.scrollIntoView({ behavior: 'smooth', block: 'nearest' });

  rewrites.push({ latex, status });

}

 

function clearRewrites() {

  document.getElementById('rewrites').innerHTML = '';

  rewrites = [];

}

 

function showFb(msg, type) {

  const el = document.getElementById('fb');

  const t = {

    correct: ['var(--color-background-success)', 'var(--color-text-success)', 'var(--color-border-success)'],

    error:   ['var(--color-background-danger)',  'var(--color-text-danger)',  'var(--color-border-danger)'],

    info:    ['var(--color-background-info)',    'var(--color-text-info)',    'var(--color-border-info)']

  };

  const [bg, tc, bc] = t[type] || t.info;

  el.style.cssText = `display:block;margin-top:10px;padding:12px 16px;border-radius:var(--border-radius-md);font-size:14px;line-height:1.6;background:${bg};color:${tc};border:0.5px solid ${bc};`;

  el.innerHTML = msg;

}

function hideFb() { document.getElementById('fb').style.display = 'none'; }

function upd() {

  document.getElementById('sc').textContent = score;

  document.getElementById('st').textContent = streak;

}

 

/* ---- Mode + question lifecycle ---- */

function setMode(m) {

  mode = m;

  ['simplify','explore'].forEach(id =>

    document.getElementById('t-'+id).className = 'tab' + (id === m ? ' on' : ''));

  document.getElementById('btn-hint').style.display = m === 'simplify' ? 'inline-block' : 'none';

  nxtQ();

}

 

function nxtQ() {

  clearRewrites();

  hideFb();

  Q = genQ();

  const pd = document.getElementById('problem-display');

  pd.classList.remove('loading');

  pd.innerHTML = renderLatex(Q.problemLatex);

  document.getElementById('inst').textContent = mode === 'simplify'

    ? 'Rewrite this expression step by step — you can enter intermediate forms and work towards the fully simplified result.'

    : 'Enter as many different equivalent forms of this expression as you can. Each will be checked and logged.';

  const input = document.getElementById('student-input');

  if (input && 'value' in input) input.value = '';

  if (input && input.focus) setTimeout(() => input.focus(), 60);

}

 

/* ---- Check student's input ---- */

function chk() {

  if (!Q) return;

  const input = document.getElementById('student-input');

  const studentLatex = (input.value || '').trim();

  if (!studentLatex) {

    showFb('Please enter your rewrite.', 'info');

    return;

  }

 

  // Duplicate of a previous rewrite?

  if (rewrites.some(r => canonicalize(r.latex) === canonicalize(studentLatex))) {

    showFb('You already entered this form. Try another.', 'info');

    return;

  }

 

  // Same as original?

  if (canonicalize(studentLatex) === canonicalize(Q.problemLatex)) {

    showFb('That is the original expression — try a different form.', 'info');

    return;

  }

 

  const eq = checkEquivalent(Q.problemLatex, studentLatex);

 

  if (!eq.ok) {

    addRewrite(studentLatex, 'incorrect');

    streak = 0;

    upd();

    showFb('Not equivalent. Check your algebra.', 'error');

    input.value = '';

    input.focus();

    return;

  }

 

  const isSimp = isSimplifiedForm(studentLatex, Q.simplifiedLatex);

  addRewrite(studentLatex, 'correct');

  score++;

  streak++;

  upd();

 

  if (mode === 'simplify') {

    if (isSimp) {

      showFb('Fully simplified! Press "New question" to continue.', 'correct');

    } else {

      showFb('Equivalent — but still reducible. Try simplifying further.', 'info');

    }

  } else {

    let msg = `Equivalent! ${rewrites.filter(r => r.status === 'correct').length} correct form${rewrites.filter(r => r.status === 'correct').length !== 1 ? 's' : ''} so far.`;

    if (isSimp) msg += ' This is the fully simplified form.';

    showFb(msg, 'correct');

  }

 

  input.value = '';

  input.focus();

}

 

function hint() {

  if (!Q) return;

  showFb('Hint: ' + Q.hint.replace(/\$([^$]+)\$/g, (_, tex) => renderLatex(tex)), 'info');

}

 

function showA() {

  if (!Q) return;

  streak = 0;

  upd();

  // Add the simplified form as a rewrite entry (without removing existing rewrites)

  const already = rewrites.some(r => canonicalize(r.latex) === canonicalize(Q.simplifiedLatex));

  if (!already) {

    addRewrite(Q.simplifiedLatex, 'answer');

  }

  showFb('The fully simplified form is shown above.', 'info');

}

        chk();

      }

    });

  }

 

  setMode('simplify');

}
